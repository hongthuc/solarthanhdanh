    // 1. Tự động chuyển đổi nguồn ảnh dự phòng nếu Google Drive giới hạn băng thông
    function handleDriveImgError(imgEl, labelText) {
      const fallbackUrl = imgEl.getAttribute('data-fallback');
      if (fallbackUrl && imgEl.src !== fallbackUrl) {
        imgEl.src = fallbackUrl;
        return;
      }
      const encodedText = encodeURIComponent(labelText || 'SOLAR THANH DANH');
      imgEl.onerror = null;
      imgEl.src = 'https://placehold.co/600x450/14532d/ffffff?text=' + encodedText;

    let currentBannerIndex = 0;
    const totalBannerSlides = 2;
    let bannerInterval = null;

    function renderBannerSlide(index) {
      currentBannerIndex = (index + totalBannerSlides) % totalBannerSlides;

      for (let i = 0; i < totalBannerSlides; i++) {
        const dSlide = document.getElementById('desktopSlide' + i);
        const mSlide = document.getElementById('mobileSlide' + i);
        const dDot = document.getElementById('desktopDot' + i);
        const mDot = document.getElementById('mobileDot' + i);

        if (i === currentBannerIndex) {
          if (dSlide) { dSlide.classList.remove('opacity-0', 'z-0'); dSlide.classList.add('opacity-100', 'z-10'); }
          if (mSlide) { mSlide.classList.remove('opacity-0', 'z-0'); mSlide.classList.add('opacity-100', 'z-10'); }
          if (dDot) { dDot.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all'; }
          if (mDot) { mDot.className = 'w-4 h-1.5 rounded-full bg-amber-400 transition-all'; }
        } else {
          if (dSlide) { dSlide.classList.remove('opacity-100', 'z-10'); dSlide.classList.add('opacity-0', 'z-0'); }
          if (mSlide) { mSlide.classList.remove('opacity-100', 'z-10'); mSlide.classList.add('opacity-0', 'z-0'); }
          if (dDot) { dDot.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all'; }
          if (mDot) { mDot.className = 'w-2 h-1.5 rounded-full bg-white/50 transition-all'; }
        }
      }
    }
